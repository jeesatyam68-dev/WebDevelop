       1.) Ek number even hai ya odd print karo.
         
       Answer:) function identify(a){
           if(a%2===0){
               return `${a} is an even number`
            }
            else{
                return `${a} is a odd number`
            }
        }
        console.log(identify(2))




        2.)  Teen numbers me sabse bada number print karo.

        function bigger(b, c, d){
            if(b>c && b>d){
                return `${b} is the greatest`
            }
            else if(c>b && c>d){

            }
        }
        console.log(bigger(2, 3, 5));
